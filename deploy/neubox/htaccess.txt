# Be Online — sitio estatico (Astro) + proxy mismo-origen. Reemplaza WordPress.
RewriteEngine On

# Puentes al API/mapa/SEPOMEX via _proxy.php (mismo-origen, sin CORS)
RewriteRule ^(api-proxy|map-proxy|map-api|sepomex-proxy)/ /_proxy.php [L,QSA]

DirectoryIndex index.html
ErrorDocument 404 /404.html

<IfModule mod_headers.c>
  # HTML sin cache (el host tiene AccelerateWP/cache de pagina — leccion Baxnet)
  <FilesMatch "\.html?$">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
    Header unset ETag
  </FilesMatch>
  # Assets con hash: cache larga
  <FilesMatch "\.(js|css|woff2?|ttf|otf|png|jpe?g|webp|avif|svg|ico)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>
